class Trabajadora(models.Model):
    class Area(models.TextChoices):
        CONFECCION = "CONFECCION", "Confección"
        BORDADO = "BORDADO", "Bordado"
        COMPRAS = "COMPRAS", "Compras"
        PRUEBAS = "PRUEBAS", "Pruebas"
        LAVANDERIA = "LAVANDERIA", "Lavandería"
        ACABADOS = "ACABADOS", "Acabados"
        OTRO = "OTRO", "Otro"

    nombre = models.CharField(max_length=150)
    area = models.CharField(max_length=20, choices=Area.choices)
    activa = models.BooleanField(default=True)

    # Una persona puede tener varias habilidades.
    habilidades = models.ManyToManyField(
        "Operacion",
        blank=True,
        related_name="trabajadoras_habilitadas",
    )

    observaciones = models.TextField(blank=True)

    def __str__(self):
        return f"{self.nombre} - {self.get_area_display()}"


class DisponibilidadSemanal(models.Model):
    DIAS = [
        (0, "Lunes"),
        (1, "Martes"),
        (2, "Miércoles"),
        (3, "Jueves"),
        (4, "Viernes"),
        (5, "Sábado"),
        (6, "Domingo"),
    ]

    trabajadora = models.ForeignKey(
        Trabajadora,
        on_delete=models.CASCADE,
        related_name="horarios",
    )
    dia_semana = models.PositiveSmallIntegerField(choices=DIAS)
    horas_disponibles = models.DecimalField(
        max_digits=6,
        decimal_places=2,
        default=0,
        validators=[MinValueValidator(Decimal("0"))],
    )

    class Meta:
        constraints = [
            models.UniqueConstraint(
                fields=["trabajadora", "dia_semana"],
                name="horario_unico_por_trabajadora_dia",
            )
        ]
        ordering = ["dia_semana"]

    def __str__(self):
        return (
            f"{self.trabajadora.nombre} - "
            f"{self.get_dia_semana_display()}: "
            f"{self.horas_disponibles} h"
        )

class TipoVestido(models.Model):
    nombre = models.CharField(max_length=100, unique=True)
    activo = models.BooleanField(default=True)
    descripcion = models.TextField(blank=True)

    def __str__(self):
        return self.nombre


class Operacion(models.Model):
    nombre = models.CharField(max_length=120, unique=True)
    area_responsable = models.CharField(
        max_length=20,
        choices=Trabajadora.Area.choices,
        default=Trabajadora.Area.OTRO,
    )
    descripcion = models.TextField(blank=True)
    activa = models.BooleanField(default=True)

class RutaProduccion(models.Model):
    class Tipo(models.TextChoices):
        TALLER_A = "A", "Taller A - Confección desde cero"
        TALLER_C = "C", "Taller C - Compra y modificación"

    tipo_vestido = models.ForeignKey(
        TipoVestido,
        on_delete=models.PROTECT,
        related_name="rutas",
    )
    taller = models.CharField(max_length=1, choices=Tipo.choices)
    activa = models.BooleanField(default=True)
    descripcion = models.TextField(blank=True)

    class Meta:
        constraints = [
            models.UniqueConstraint(
                fields=["tipo_vestido", "taller"],
                name="ruta_unica_por_tipo_y_taller",
            )
        ]

    def __str__(self):
        return f"{self.tipo_vestido} - Taller {self.taller}"


class PlantillaOperacion(models.Model):
    """
    Define cada etapa y sus tiempos editables para un vestido y ruta.

    TRABAJO: consume horas de una persona o recurso.
    ESPERA: transcurre tiempo, pero no consume horas de personal.
    """

    class TipoTiempo(models.TextChoices):
        TRABAJO = "TRABAJO", "Trabajo efectivo"
        ESPERA = "ESPERA", "Espera o plazo de proveedor"

    ruta = models.ForeignKey(
        RutaProduccion,
        on_delete=models.CASCADE,
        related_name="operaciones",
    )
    operacion = models.ForeignKey(
        Operacion,
        on_delete=models.PROTECT,
        related_name="plantillas",
    )
    orden = models.PositiveIntegerField(default=1)

    tipo_tiempo = models.CharField(
        max_length=10,
        choices=TipoTiempo.choices,
        default=TipoTiempo.TRABAJO,
    )

    # Cada proceso tiene su propio tiempo, modificable.
    horas_estimadas = models.DecimalField(
        max_digits=9,
        decimal_places=2,
        validators=[MinValueValidator(Decimal("0"))],
    )

    obligatoria = models.BooleanField(default=True)
    activa = models.BooleanField(default=True)

    # Permite indicar que una etapa debe esperar a otra.
    depende_de = models.ManyToManyField(
        "self",
        symmetrical=False,
        blank=True,
        related_name="operaciones_siguientes",
    )

    observaciones = models.TextField(blank=True)

    class Meta:
        ordering = ["orden", "id"]
        constraints = [
            models.UniqueConstraint(
                fields=["ruta", "operacion"],
                name="operacion_unica_por_ruta",
            )
        ]

    def __str__(self):
        return (
            f"{self.ruta} - {self.operacion}: "
            f"{self.horas_estimadas} h"
        )

class Contrato(models.Model):
    class Estado(models.TextChoices):
        ACTIVO = "ACTIVO", "Activo"
        EN_PRODUCCION = "EN_PRODUCCION", "En producción"
        ENTREGADO = "ENTREGADO", "Entregado"
        CANCELADO = "CANCELADO", "Cancelado"

    simulacion = models.OneToOneField(
        SimulacionContrato,
        on_delete=models.PROTECT,
        related_name="contrato",
    )
    numero = models.CharField(max_length=40, unique=True)
    fecha_firma = models.DateField(default=timezone.localdate)
    fecha_entrega = models.DateField()
    precio_total = models.DecimalField(
        max_digits=12,
        decimal_places=2,
        validators=[MinValueValidator(Decimal("0"))],
    )
    estado = models.CharField(
        max_length=15,
        choices=Estado.choices,
        default=Estado.ACTIVO,
    )
    condiciones_pago = models.TextField(blank=True)
    observaciones = models.TextField(blank=True)

    @property
    def total_pagado(self):
        return self.pagos.filter(confirmado=True).aggregate(
            total=Sum("importe")
        )["total"] or Decimal("0.00")

    @property
    def saldo_pendiente(self):
        return self.precio_total - self.total_pagado

    def __str__(self):
        return self.numero

class Pago(models.Model):
    class Metodo(models.TextChoices):
        EFECTIVO = "EFECTIVO", "Efectivo"
        TRANSFERENCIA = "TRANSFERENCIA", "Transferencia"
        TARJETA = "TARJETA", "Tarjeta"
        OTRO = "OTRO", "Otro"

    contrato = models.ForeignKey(
        Contrato,
        on_delete=models.PROTECT,
        related_name="pagos",
    )
    fecha = models.DateField(default=timezone.localdate)
    importe = models.DecimalField(
        max_digits=12,
        decimal_places=2,
        validators=[MinValueValidator(Decimal("0.01"))],
    )
    metodo = models.CharField(max_length=20, choices=Metodo.choices)
    concepto = models.CharField(max_length=150, default="Pago de contrato")
    confirmado = models.BooleanField(default=False)
    referencia = models.CharField(max_length=100, blank=True)
    registrado_por = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
    )

    def save(self, *args, **kwargs):
        """
        Al confirmar un pago, crea o actualiza su movimiento financiero.
        Si no está confirmado, no se contabiliza como ingreso.
        """
        with transaction.atomic():
            super().save(*args, **kwargs)

            if self.confirmado:
                categoria, _ = CategoriaFinanciera.objects.get_or_create(
                    nombre="Pagos de contratos",
                    defaults={
                        "tipo": CategoriaFinanciera.Tipo.INGRESO
                    },
                )

                MovimientoFinanciero.objects.update_or_create(
                    pago=self,
                    defaults={
                        "fecha": self.fecha,
                        "tipo": MovimientoFinanciero.Tipo.INGRESO,
                        "categoria": categoria,
                        "importe": self.importe,
                        "descripcion": (
                            f"{self.concepto} - Contrato "
                            f"{self.contrato.numero}"
                        ),
                    },
                )
            else:
                MovimientoFinanciero.objects.filter(
                    pago=self
                ).delete()

    def __str__(self):
        return f"Pago de {self.importe} - {self.contrato.numero}"


class MovimientoFinanciero(models.Model):
    class Tipo(models.TextChoices):
        INGRESO = "INGRESO", "Ingreso"
        EGRESO = "EGRESO", "Egreso"

    fecha = models.DateField(default=timezone.localdate)
    tipo = models.CharField(max_length=10, choices=Tipo.choices)
    categoria = models.ForeignKey(
        CategoriaFinanciera,
        on_delete=models.PROTECT,
        related_name="movimientos",
    )
    importe = models.DecimalField(
        max_digits=12,
        decimal_places=2,
        validators=[MinValueValidator(Decimal("0.01"))],
    )
    descripcion = models.CharField(max_length=250)

    # Un pago confirmado genera un único movimiento asociado.
    pago = models.OneToOneField(
        Pago,
        on_delete=models.PROTECT,
        null=True,
        blank=True,
        related_name="movimiento_financiero",
    )

    contrato = models.ForeignKey(
        Contrato,
        on_delete=models.PROTECT,
        null=True,
        blank=True,
        related_name="movimientos_financieros",
    )

    creado_en = models.DateTimeField(auto_now_add=True)

    class Meta:
        ordering = ["-fecha", "-id"]

    def __str__(self):
        return f"{self.tipo}: {self.importe} - {self.fecha}"

def cargar_catalogo_inicial():
    """
    Ejecutar una vez desde Django shell.
    Crea los vestidos, las operaciones, las dos rutas y sus tiempos.

    No sobrescribe las horas que ya hayas modificado posteriormente.
    """
    with transaction.atomic():
        for nombre_vestido, procesos in DATOS_TIEMPOS.items():
            vestido, _ = TipoVestido.objects.get_or_create(
                nombre=nombre_vestido
            )

            for taller in ["A", "C"]:
                ruta, _ = RutaProduccion.objects.get_or_create(
                    tipo_vestido=vestido,
                    taller=taller,
                )

                for orden, (nombre, tiempos) in enumerate(
                    procesos.items(), start=1
                ):
                    horas = tiempos[0] if taller == "A" else tiempos[1]

                    # En el Excel, una celda vacía significa que
                    # la operación no corresponde a esa ruta.
                    if horas is None:
                        continue

                    nombre_normalizado = nombre.strip().lower()

                    area = Trabajadora.Area.OTRO
                    if nombre_normalizado == "compra materiales":
                        area = Trabajadora.Area.COMPRAS
                    elif nombre_normalizado in (
                        "patronaje", "corte de tela", "confección", "entalle"
                    ):
                        area = Trabajadora.Area.CONFECCION
                    elif nombre_normalizado == "bordado":
                        area = Trabajadora.Area.BORDADO
                    elif nombre_normalizado.startswith("prueba"):
                        area = Trabajadora.Area.PRUEBAS
                    elif nombre_normalizado == "lavandería":
                        area = Trabajadora.Area.LAVANDERIA
                    elif nombre_normalizado in ("acabados", "basta"):
                        area = Trabajadora.Area.ACABADOS

                    operacion, _ = Operacion.objects.get_or_create(
                        nombre=nombre,
                        defaults={"area_responsable": area},
                    )

                    tipo_tiempo = (
                        PlantillaOperacion.TipoTiempo.ESPERA
                        if taller == "C"
                        and nombre_normalizado == "compra materiales"
                        else PlantillaOperacion.TipoTiempo.TRABAJO
                    )

                    plantilla, creada = (
                        PlantillaOperacion.objects.get_or_create(
                            ruta=ruta,
                            operacion=operacion,
                            defaults={
                                "orden": orden,
                                "horas_estimadas": Decimal(str(horas)),
                                "tipo_tiempo": tipo_tiempo,
                                "activa": True,
                            },
                        )
                    )

                    # Si la plantilla ya existe, no se reemplaza
                    # su tiempo editable.
                    if creada:
                        plantilla.save()

        on_delete=models.PROTECT,

        "Trabajadora",
        on_delete=models.PROTECT,
        related_name="reservas_produccion",

        "PlantillaOperacion",
        on_delete=models.PROTECT,

        max_digits=8,
        decimal_places=2,

    horas_realizadas = models.DecimalField(
        max_digits=8,
        decimal_places=2,
        default=Decimal("0"),
        validators=[MinValueValidator(Decimal("0"))],
    )

    estado = models.CharField(

        default=Estado.RESERVADA,
    )
    observaciones = models.TextField(blank=True)
    creada_en = models.DateTimeField(auto_now_add=True)

    class Meta:
        ordering = ["fecha", "trabajadora", "plantilla"]
        constraints = [
            models.UniqueConstraint(
                fields=[
                    "contrato",
                    "trabajadora",
                    "plantilla",
                    "fecha",
                ],
                name="reserva_unica_por_contrato_persona_operacion_dia",
            )
        ]
        indexes = [
            models.Index(
                fields=["trabajadora", "fecha", "estado"],
                name="res_persona_fecha_estado_idx",
            ),
            models.Index(
                fields=["contrato", "estado"],
                name="reserva_contrato_estado_idx",
            ),
        ]

    @property
    def horas_pendientes(self):
        return max(
            Decimal("0"),
            self.horas_reservadas - self.horas_realizadas,
        )

    def __str__(self):
        return (
            f"{self.contrato.numero} | "
            f"{self.trabajadora.nombre} | "
            f"{self.plantilla.operacion.nombre} | "
            f"{self.fecha} | {self.horas_reservadas} h"
        )

class EsGerencia(BasePermission):
    """
    Acceso exclusivo para superusuarios o miembros
    del grupo Gerencia de Django.
    """

    message = "No tienes permisos de Gerencia."

    def has_permission(self, request, view):
        usuario = request.user

        return bool(
            usuario
            and usuario.is_authenticated
            and (
                usuario.is_superuser
                or usuario.groups.filter(
                    name="Gerencia"
                ).exists()
            )
        )


class TrabajadoraViewSet(viewsets.ModelViewSet):
    """
    CRUD de personal:
    GET, POST, PATCH, PUT y DELETE.

    DELETE está deshabilitado para conservar
    el historial del taller.
    """

    serializer_class = TrabajadoraSerializer
    permission_classes = [IsAuthenticated, EsGerencia]

    queryset = Trabajadora.objects.prefetch_related(
        "habilidades",
        "horarios",
    ).all().order_by("nombre", "id")

    def get_queryset(self):
        queryset = super().get_queryset()

        nombre = self.request.query_params.get("buscar")
        area = self.request.query_params.get("area")
        activa = self.request.query_params.get("activa")

        if nombre:
            queryset = queryset.filter(
                nombre__icontains=nombre.strip()
            )

        if area:
            queryset = queryset.filter(area=area)

        if activa is not None:
            if activa.lower() == "true":
                queryset = queryset.filter(activa=True)
            elif activa.lower() == "false":
                queryset = queryset.filter(activa=False)

        return queryset

    def perform_create(self, serializer):
        """
        Al registrar personal, crea sus siete días
        de disponibilidad con cero horas.
        Gerencia podrá configurarlos después.
        """
        with transaction.atomic():
            trabajadora = serializer.save()

            DisponibilidadSemanal.objects.bulk_create([
                DisponibilidadSemanal(
                    trabajadora=trabajadora,
                    dia_semana=dia,
                    horas_disponibles=Decimal("0"),
                )
                for dia in range(7)
            ])

    def destroy(self, request, *args, **kwargs):
        return Response(
            {
                "detalle": (
                    "No se permite eliminar personal. "
                    "Cambia su estado a inactivo."
                )
            },
            status=status.HTTP_405_METHOD_NOT_ALLOWED,
        )

    @action(detail=False, methods=["get"])
    def opciones(self, request):
        """
        Opciones para los selectores del frontend.
        """
        operaciones = Operacion.objects.filter(
            activa=True
        ).order_by("area_responsable", "nombre")

        return Response({

                {"valor": valor, "nombre": nombre}

    @action(
        detail=True,
        methods=["get", "put"],
        url_path="horarios",
    )
    def horarios(self, request, pk=None):
        trabajadora = self.get_object()

        if request.method == "GET":
            horarios = trabajadora.horarios.all()

            return Response({
                "trabajadora": trabajadora.id,
                "nombre": trabajadora.nombre,
                "horarios": DisponibilidadSerializer(
                    horarios,
                    many=True,
                ).data,
            })

        serializer = HorarioSemanalEntradaSerializer(
            data=request.data
        )
        serializer.is_valid(raise_exception=True)

        with transaction.atomic():
            for item in serializer.validated_data["horarios"]:
                DisponibilidadSemanal.objects.update_or_create(
                    trabajadora=trabajadora,
                    dia_semana=item["dia_semana"],
                    defaults={
                        "horas_disponibles":
                            item["horas_disponibles"]
                    },
                )

        horarios = trabajadora.horarios.all()

        return Response({
            "mensaje": "Horario actualizado correctamente.",
            "horarios": DisponibilidadSerializer(
                horarios,
                many=True,
            ).data,
        })

    @action(
        detail=True,
        methods=["get"],
        url_path="capacidad",
    )
    def capacidad(self, request, pk=None):
        """
        Resumen de una semana.

        Parámetro opcional:
        ?fecha=2026-10-26

        La semana se calcula de lunes a domingo.

        horas_no_disponibles se interpreta como
        descuento diario en cada fecha de ausencia.

        Las reservas COMPLETADAS se cuentan por sus
        horas reservadas para mostrar ocupación
        planificada, no solamente trabajo pendiente.
        """
        trabajadora = self.get_object()

        fecha_texto = request.query_params.get("fecha")

        if fecha_texto:
            try:
                from datetime import date
                fecha = date.fromisoformat(fecha_texto)
            except ValueError:
                return Response(
                    {"fecha": "Usa el formato AAAA-MM-DD."},
                    status=status.HTTP_400_BAD_REQUEST,
                )
        else:
            fecha = timezone.localdate()

        lunes = fecha - timedelta(days=fecha.weekday())
        domingo = lunes + timedelta(days=6)

        horarios = {
            horario.dia_semana: horario.horas_disponibles
            for horario in trabajadora.horarios.all()
        }

        ausencias = AusenciaTrabajadora.objects.filter(
            trabajadora=trabajadora,
            fecha_inicio__lte=domingo,
            fecha_fin__gte=lunes,
        )

        reservas = ReservaOperacion.objects.filter(
            trabajadora=trabajadora,
            fecha__range=(lunes, domingo),
        ).exclude(
            estado=ReservaOperacion.Estado.CANCELADA
        )

        reservas_por_fecha = {}

        for reserva in reservas:
            reservas_por_fecha.setdefault(
                reserva.fecha, Decimal("0")
            )
            reservas_por_fecha[reserva.fecha] += (
                reserva.horas_reservadas
            )

        detalle = []
        total_capacidad = Decimal("0")
        total_reservado = Decimal("0")
        total_libre = Decimal("0")

        for numero_dia in range(7):
            dia = lunes + timedelta(days=numero_dia)

            horas_base = (
                horarios.get(numero_dia, Decimal("0"))
                if trabajadora.activa
                else Decimal("0")
            )

            descuento = sum(
                (
                    ausencia.horas_no_disponibles
                    for ausencia in ausencias
                    if ausencia.fecha_inicio <= dia
                    <= ausencia.fecha_fin
                ),
                Decimal("0"),
            )

            capacidad = max(
                Decimal("0"),
                horas_base - descuento,
            )

            reservado = reservas_por_fecha.get(
                dia, Decimal("0")
            )

            libre = max(
                Decimal("0"),
                capacidad - reservado,
            )

            total_capacidad += capacidad
            total_reservado += reservado
            total_libre += libre

            detalle.append({
                "fecha": dia.isoformat(),
                "dia_semana": numero_dia,
                "horas_base": horas_base,
                "horas_ausencia": descuento,
                "horas_disponibles": capacidad,
                "horas_reservadas": reservado,
                "horas_libres": libre,
                "sobrecarga": max(
                    Decimal("0"),
                    reservado - capacidad,
                ),
            })

        ocupacion = (
            (total_reservado / total_capacidad * 100)
            if total_capacidad > 0
            else Decimal("0")
        )

        return Response({
            "trabajadora": trabajadora.id,
            "nombre": trabajadora.nombre,
            "activa": trabajadora.activa,
            "semana_inicio": lunes.isoformat(),
            "semana_fin": domingo.isoformat(),
            "horas_disponibles": total_capacidad,
            "horas_reservadas": total_reservado,
            "horas_libres": total_libre,
            "ocupacion_porcentaje": round(ocupacion, 2),
            "detalle": detalle,
        })
